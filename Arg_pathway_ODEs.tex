% --- pathway intermediates -----------------------------------------------

\begin{align}
\frac{d[\mathrm{Glu}]}{dt}
  &= \xi \Biggl[
      \underbrace{
        V_{\max}^{ghdA}
        \cdot\frac{[\alpha\mathrm{KG}]}{K_m^{ghdA,\,\alpha\mathrm{KG}}+[\alpha\mathrm{KG}]}
        \cdot\frac{[\mathrm{NH}_4^+]}{K_m^{ghdA,\,\mathrm{NH}_4}+[\mathrm{NH}_4^+]}
        \cdot\frac{[\mathrm{NADPH}]}{K_m^{ghdA,\,\mathrm{NADPH}}+[\mathrm{NADPH}]}
      }_{v_{ghdA}}
  \notag\\
  &\qquad\qquad -
      \underbrace{
        \frac{V_{\max}^{argJ}\,[\mathrm{AcOrn}][\mathrm{Glu}]}
             {[\mathrm{AcOrn}][\mathrm{Glu}]
              +K_m^{argJ,\,\mathrm{Glu}}[\mathrm{AcOrn}]
               \!\left(1+\dfrac{[\mathrm{Orn}]}{K_i^{argJ,\,\mathrm{Orn}}}\right)
              +K_m^{argJ,\,\mathrm{AcOrn}}[\mathrm{Glu}]}
      }_{v_{argJ}}
    \Biggr]
\label{eq:dGlu}
\end{align}

\begin{align}
